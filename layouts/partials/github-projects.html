{{/* Fetches personal GitHub repos for site.Params.github_username at build time. Only includes repos named in site.Params.github_projects_include (whitelist order). Returns a slice of GitHub repo objects. */}} {{ $username := site.Params.github_username }} {{ $include := site.Params.github_projects_include | default (slice) }} {{ $filtered := slice }} {{ if not $include }} {{ warnf "params.github_projects_include is empty; no projects will be shown." }} {{ else if $username }} {{ $url := printf "https://api.github.com/users/%s/repos?type=owner&sort=updated&per_page=100" $username }} {{ $opts := dict "headers" (dict "Accept" "application/vnd.github+json" "User-Agent" (printf "hugo-portfolio-%s" $username) ) }} {{ with try (resources.GetRemote $url $opts) }} {{ with .Err }} {{ warnf "Failed to fetch GitHub repos: %s" . }} {{ else with .Value }} {{ $repos := . | transform.Unmarshal }} {{ range $name := $include }} {{ $match := false }} {{ range $repos }} {{ if and (eq .name $name) (not .fork) (not .archived) }} {{ $filtered = $filtered | append . }} {{ $match = true }} {{ end }} {{ end }} {{ if not $match }} {{ warnf "Whitelisted GitHub project %q not found (or is a fork/archived)." $name }} {{ end }} {{ end }} {{ end }} {{ end }} {{ else }} {{ warnf "params.github_username is not set; cannot fetch GitHub projects." }} {{ end }} {{ return $filtered }}
